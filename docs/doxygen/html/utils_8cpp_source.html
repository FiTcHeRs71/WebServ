109bool is_valid_ipv4(const string &host)
110{
111 size_t start = 0;
112 size_t i = 0;
113 size_t dot;
114
115 if (count(host.begin(), host.end(), '.') != 3)
116 return (false);
117 while (i < 4)
118 {
119 dot = host.find('.', start);
120 string segment = host.substr(start, dot - start);
121
122 if (segment.empty() || segment.size() > 3)
123 return (false);
124 if (segment.find_first_not_of("0123456789") != string::npos)
125 return (false);
126 if (segment.size() > 1 && segment[0] == '0')
127 return (false);
128 if (atoi(segment.c_str()) > 255)
129 return (false);
130 start = dot + 1;
131 i++;
132 }
133 return (true);
134}

144size_t parse_body_size(const string &value)
145{
146 int multiplier;
147 long size_converted;
148 char* p_end = NULL;
149 errno = 0;
150
151 if (value.empty())
152 throw runtime_error(value + " is not a valid body size");
153 size_converted = strtol(value.c_str(), &p_end, 10);
154 if (p_end == value.c_str() || size_converted < 0 || errno == ERANGE)
155 throw runtime_error(value + " is not a valid body size");
156 if (*p_end == '\0')
157 multiplier = 1;
158 else if ((*p_end == 'K' || *p_end == 'k') && p_end[1] == '\0')
159 multiplier = 1024;
160 else if ((*p_end == 'M' || *p_end == 'm') && p_end[1] == '\0')
161 multiplier = 1024 * 1024;
162 else
163 throw runtime_error(value + " is not a valid body size");
164 if (size_converted > INT_MAX / multiplier)
165 throw runtime_error(value + " is a too big body size");
166 return (static_cast<size_t>(size_converted * multiplier));
167}
