220Response Response::BuildError(int code, const ServerConfig &server)
221{
222 Response res;
223 res.SetStatus(code);
224 res.SetHeader("Content-Type", "text/html");
225 const map<int, string> &ErrorPages = server.getErrorPages();
226 map<int, string>::const_iterator it = ErrorPages.find(code);
227 if (it == ErrorPages.end())
228 {
229 res.generateBuiltInError();
230 return res;
231 }
232 const LocationConfig *loc = server.Resolve(it->second);
233 if(!loc)
234 {
235 res.generateBuiltInError();
236 return res;
237 }
238 string ErrorPath = server.build_path(*loc, it->second);
239 if(!ErrorPath.empty())
240 {
241 int fd = open(ErrorPath.c_str(), O_RDONLY);
242 if (fd < 0)
243 {
244 res.generateBuiltInError();
245 return res;
246 }
247 char buf[4096];
248 ssize_t n;
249 string body;
250 while ((n = read(fd, buf, sizeof(buf))) > 0)
251 {
252 body.append(buf, static_cast<size_t>(n));
253 }
254 close(fd);
255 if (n < 0 || body.empty())
256 {
257 res.generateBuiltInError();
258 return res;
259 }
260 else
261 res.SetBody(body);
262 }
263 else
264 res.generateBuiltInError();
