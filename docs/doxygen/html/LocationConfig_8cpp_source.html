213void LocationConfig::parse_location(vector<string> &token, size_t &i)
214{
215 set<string> seen;
216 string CgiExt;
217
218 this->_Path = token[i];
219 i += 2; // saute le PATH + "{"
220 while (i < token.size() && token[i] != "}")
221 {
222 string key = token[i];
223 i++;
224
225 if (!seen.insert(key).second && key != "cgi_ext" && key != "cgi_pass")
226 throw runtime_error("Multiple definition of " + key + " not allowed in same location blocks");
227
228 vector<string> value = collect_values(token, i);
229 if (key == "allow_methods")
230 this->_Methods = parse_allow_methods(value);
231 else if (key == "root")
232 this->_Root = parse_root(value);
233 else if (key == "index")
234 this->_Index = parse_index(value);
235 else if (key == "autoindex")
236 this->_AutoIndex = parse_auto_index(value);
237 else if (key == "cgi_ext")
238 {
239 if (!CgiExt.empty())
240 throw runtime_error("Consecutive cgi_ext without any cgi_pass");
241 CgiExt = parse_cgi_ext(value);
242 }
243 else if (key == "cgi_pass")
244 {
245 if (CgiExt.empty())
246 throw runtime_error("No cgi_ext preceding cgi_pass");
247 else if (_Cgi.count(CgiExt))
248 throw runtime_error("Multiple definition for the same cgi_ext");
249 _Cgi[CgiExt] = parse_cgi_pass(value);
250 CgiExt.clear();
251 }
252 else if (key == "client_max_body_size")
253 {
254 if (value.size() != 1)
255 throw runtime_error(key + " needs only one arguments");
256 this->_HasClientMaxBodySize = true;
257 this->_ClientMaxBodySize = parse_body_size(value[0]);
258 }
259 else if (key == "return")
260 {
261 if (value.empty() || value.size() > 2)
262 throw runtime_error(key + " needs one or two arguments");
263 this->_HasReturn = true;
264 this->_ReturnCode = parser_return_code(value[0], value.size());
265 if (value.size() == 2)
266 this->_ReturnTarget = value[1];
267 }
268 else if (key == "upload_store")
269 {
270 this->_UploadStore = parse_upload_store(value);
271 this->_HasUploadStore = true;
272 }
273 else
274 throw runtime_error(key + " is not a valid instructions in location bloc");
275 }
276 if (!CgiExt.empty())
277 throw runtime_error("cgi_pass is mandatory with a cgi_ext key");
278 i++; // saute le "}" avant de rendre le i aparse bloc server
279}
