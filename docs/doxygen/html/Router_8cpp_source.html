82static Response serveFile(const ServerConfig &server, string file)
83{
84 Response res;
85 map<string, string> mime = mime_table();
86 int fd;
87 if ((fd = open(file.c_str(), O_RDONLY)) < 0)
88 return(Response::BuildError(403, server));
89 char buf[4096];
90 ssize_t n;
91 string body;
92 while ((n = read(fd, buf, sizeof(buf))) > 0)
93 body.append(buf, static_cast<size_t>(n));
94 close(fd);
95 if (n < 0)
96 return(Response::BuildError(403, server));
97 getMime(res, mime, file);
98 res.SetStatus(200);
99 if (n == 0 && body.empty())
100 res.SetBody("");
101 else
102 res.SetBody(body);
103 return (res);
104}

209static string normalizePath(const string &strIn, bool &escaped)
210{
211 size_t i = 0;
212 bool abs = !strIn.empty() && strIn[0] == '/';
213 vector<string> st;
214
215 escaped = false;
216 while (i < strIn.size())
217 {
218 while(i < strIn.size() && strIn[i] == '/')
219 i++;
220 if(i >= strIn.size())
221 break ;
222 size_t j = strIn.find('/', i);
223 if (j == string::npos)
224 j = strIn.size();
225 string part = strIn.substr(i, j - i);
226 i = j;
227 if (part == "." || part.empty())
228 continue ;
229 else if (part == "..")
230 {
231 if (st.empty())
232 {
233 escaped = true;
234 return("");
235 }
236 st.pop_back();
237 continue ;
238 }
239 st.push_back(part);
240 }
241 string out = abs ? "/" : "";
242 for(size_t k = 0; k < st.size(); k++)
243 {
244 if (!out.empty() && out[out.size() - 1] != '/')
245 out += "/";
246 out += st[k];
247 }
248 if (abs && st.empty())
249 out = "/";
250 return (out);

303bool parse_multipart(const std::string &body, const std::string &boundary,
304 vector<TMultipartPart> &out)
305{
306 if (body.empty() || boundary.empty())
307 return false;
308 string delimiter = "--" + boundary;
309 size_t i = body.find(delimiter);
310 if (i == string::npos)
311 return false;
312 while(i < body.size())
313 {
314 TMultipartPart part;
315 size_t hdrsEnd;
316 size_t dataEnd;
317
318 i += delimiter.size();
319 if (i + 1 < body.size() && body[i] == '-' && body[i + 1] == '-')
320 return true;
321 if (i + 1 >= body.size() || body[i] != '\r' || body[i + 1] != '\n')
322 return false;
323 hdrsEnd = body.find("\r\n\r\n", i);
324 if (hdrsEnd == string::npos)
325 return false;
326 part.Name = findParam(body.substr(i, hdrsEnd - i), "name=");
327 part.Filename = findParam(body.substr(i, hdrsEnd - i), "filename=");
328 part.ContentType = findParam(body.substr(i, hdrsEnd - i), "Content-Type: ");
329 i = hdrsEnd + 4;
330 dataEnd = body.find("\r\n" + delimiter, i);
331 if (dataEnd == string::npos)
332 return false;
333 part.Data = body.substr(i, dataEnd - i);
334 out.push_back(part);
335 i = dataEnd + 2;
336 }
337 return false;
338}

490static Response dispatch(const Request &request, const ServerConfig &server, Connection &connection)
491{
492 const LocationConfig *loc = server.Resolve(request.getPath());
493 if (!loc)
494 return(Response::BuildError(404, server));
495 else if (loc->hasReturn())
496 return(serveReturn(server, *loc));
497 else if (loc->getMethods().count(request.getMethod()) == 0)
498 {
499 Response response = Response::BuildError(405, server);
500 response.SetHeader("Allow", buildAllowHeader(*loc));
501 return (response);
502 }
503 else if (request.getPath() == "/session" || request.getPath() == "/session/")
504 return (handleSession(request, server));
505 string file = server.build_path(*loc, request.getPath());
506 if (file.empty())
507 return (Response::BuildError(500, server));
508 else if (!isInsideRoot(loc->getRoot(), file))
509 return (Response::BuildError(403, server));
510 else if (request.getMethod() == "DELETE")
511 return (handleDelete(server, file));
512 else if (isCgi(request, *loc))
513 {
514 CgiProcess &cgi = connection.getCgi();
515 const ConfigParser *config = request.getConfigParser();
516
517 if (config == NULL)
518 return (Response::BuildError(502, server));
519 if (!cgi.Start(request, *loc, server, connection, *config, file))
520 return (Response::BuildError(502, server));
521 return (Response());
522 }
523 else if (request.getMethod() == "POST")
524 {
525 if (loc->hasUploadStore())
526 {
527 return (handleUpload(request, server, *loc));
528 }
529 else
530 {
531 Response resp;
532
533 resp.SetStatus(200);
534 resp.SetBody("");
535 return resp;
536 }
537 }
538 else
539 {
540 struct stat statbuf;
541 if (stat(file.c_str(), &statbuf) < 0)
542 return(Response::BuildError(404, server));
543 if (S_ISDIR(statbuf.st_mode))
544 {
545 return (serveDir(request, *loc, server, file));
546 }
547 else if (S_ISREG(statbuf.st_mode))
548 {
549 return (serveFile(server, file));
550 }
551 else
552 return(Response::BuildError(404, server));
553 }
554}
